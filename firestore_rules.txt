rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Reglas para usuarios (incluye datos de conductor)
    match /users/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || request.auth.token.driver == true);
      allow write: if request.auth != null && request.auth.uid == uid;
    }
    
    // Reglas para viajes
  match /rides/{rideId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null; // o restringe a rider/driver según tu modelo
    }
  }
}
